       1. SYNTHESIZED "DIP-DIP" (LUB-DUB) HEARTBEAT AUDIO (Web Audio API)

    let audioCtx = null;
    let heartbeatAudioEnabled = true;
    let heartbeatInterval = null;

    function initAudioContext() {
      if (!audioCtx) {
        const AudioContextClass = window.AudioContext || window.webkitAudioContext;
        if (AudioContextClass) {
          audioCtx = new AudioContextClass();
        }
      }
      if (audioCtx && audioCtx.state === 'suspended') {
        audioCtx.resume();
      }
    }

    // Acoustic stethoscope synthesis for the two cardiac components: "Lub" & "Dub"
    function playSingleThump(time, startFreq, duration, volume) {
      if (!audioCtx) return;
      try {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        const filter = audioCtx.createBiquadFilter();

        // Lowpass filter to create a warm, muffled chest-wall acoustic sound
        filter.type = 'lowpass';
        filter.frequency.setValueAtTime(140, time);

        // Sub-bass pitch sweep
        osc.type = 'sine';
        osc.frequency.setValueAtTime(startFreq, time);
        osc.frequency.exponentialRampToValueAtTime(32, time + duration);

        // Attack and decay envelope
        gain.gain.setValueAtTime(0.001, time);
        gain.gain.linearRampToValueAtTime(volume, time + 0.02);
        gain.gain.exponentialRampToValueAtTime(0.001, time + duration);

        osc.connect(filter);
        filter.connect(gain);
        gain.connect(audioCtx.destination);

        osc.start(time);
        osc.stop(time + duration);
      } catch (e) {
        // Audio error handling
      }
    }

    // Play both "dip" sounds with authentic timing
    function triggerDipDipSound() {
      if (!heartbeatAudioEnabled) return;
      initAudioContext();
      if (!audioCtx) return;

      const now = audioCtx.currentTime;
      // First thump: "Dip" (S1 - Systole)
      playSingleThump(now, 72, 0.12, 0.65);
      // Second thump: "Dip" (S2 - Diastole ~170ms later)
      playSingleThump(now + 0.17, 95, 0.10, 0.5);
    }

    const canvas = document.getElementById('boatTrackerCanvas');

    function getSplinePoint(t) {
      const n = pathPoints.length - 1;
      const index = Math.min(Math.floor(t * n), n - 1);
      const segmentT = (t * n) - index;
      const p0 = pathPoints[index];
      const p1 = pathPoints[index + 1];
      return {
        x: (p0.x + (p1.x - p0.x) * segmentT) * canvas.width,
        y: (p0.y + (p1.y - p0.y) * segmentT) * canvas.height
      };
    }

    function drawMapSimulation() {
      if (!ctx || canvas.width === 0) return;
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      // River Water Channel

      ctx.lineWidth = 42;
      ctx.strokeStyle = '#0f2744';
      ctx.lineCap = 'round';
      pathPoints.forEach((p, idx) => {
        const px = p.x * canvas.width;
        const py = p.y * canvas.height;
        if (idx === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
      });
      ctx.stroke();

      // River Center Line
      ctx.beginPath();
      ctx.lineWidth = 4;
      ctx.strokeStyle = '#0284c7';
      ctx.setLineDash([8, 12]);
      pathPoints.forEach((p, idx) => {
        const px = p.x * canvas.width;
        const py = p.y * canvas.height;

      // Boat Movement

      if (boatProgress > 0.98) boatProgress = 0.05;
      const boatPos = getSplinePoint(boatProgress);

      // Radar Sweep
      radarAngle += 0.04;
      ctx.save();
      ctx.translate(boatPos.x, boatPos.y);
      ctx.beginPath();
      ctx.strokeStyle = 'rgba(45, 212, 191, 0.3)';
      ctx.arc(0, 0, 42, 0, Math.PI * 2);
      ctx.stroke();
      ctx.beginPath();
      ctx.moveTo(0, 0);

      ctx.moveTo(0, -11);
      ctx.lineTo(7, 7);
      ctx.lineTo(-7, 7);
      ctx.closePath();
      ctx.fill();
      ctx.restore();

      requestAnimationFrame(drawMapSimulation);
    }

    function handleUserMessage(event) {
      event.preventDefault();
      const input = document.getElementById('chat-input');
      const query = input.value.trim();
      if (!query) return;

      appendChatMessage('user', query);
      input.value = '';

      const typing = document.getElementById('typing-indicator');
      typing.classList.remove('hidden');

      setTimeout(() => {
        typing.classList.add('hidden');
        generateMedicalAnswer(query);
      }, 600);
    }

    function appendChatMessage(sender, text, suggestedMeds = []) {
      const container = document.getElementById('chat-messages');
      const msgDiv = document.createElement('div');
      
      if (sender === 'user') {
        msgDiv.className = "flex items-start gap-3 justify-end";
        msgDiv.innerHTML = `
          <div class="bg-med-600 text-white p-4 rounded-2xl rounded-tr-none text-sm shadow-sm max-w-[85%] leading-relaxed">
            ${text}
          </div>
          <div class="w-8 h-8 rounded-xl bg-slate-200 text-slate-700 flex items-center justify-center shrink-0 text-xs font-bold">
            You
          </div>
        `;
      } else {
        msgDiv.className = "flex items-start gap-3 max-w-[85%]";
        let medsHtml = '';
        if (suggestedMeds && suggestedMeds.length > 0) {
          medsHtml = `
            <div class="mt-3 pt-3 border-t border-slate-100 space-y-2">
              <span class="text-[11px] font-bold text-slate-400 uppercase tracking-wider block">Recommended Relief Item:</span>
              ${suggestedMeds.map(m => `
                <div class="flex items-center justify-between p-2 rounded-xl bg-med-50/70 border border-med-200 text-xs">
                  <div class="font-medium text-slate-800">${m.name} <span class="text-med-600 font-bold">($${m.price.toFixed(2)})</span></div>
                  <button onclick="addToCart('${m.name}', ${m.price})" class="px-3 py-1 rounded-lg bg-med-600 hover:bg-med-700 text-white font-bold text-[11px] transition flex items-center gap-1 shadow-sm">
                    <i data-lucide="plus-circle" class="w-3.5 h-3.5"></i> Add to Boat Order
                  </button>
                </div>
              `).join('')}
            </div>
          `;
        }

        msgDiv.innerHTML = `
          <div class="w-8 h-8 rounded-xl bg-gradient-to-tr from-med-600 to-tealAccent-500 text-white flex items-center justify-center shrink-0 text-xs font-bold">
            <i data-lucide="cross" class="w-4 h-4"></i>
          </div>
          <div class="bg-white p-4 rounded-2xl rounded-tl-none border border-slate-200 text-sm text-slate-700 shadow-sm leading-relaxed">
            ${text}
            ${medsHtml}
          </div>
        `;
      }

      container.appendChild(msgDiv);
      container.scrollTop = container.scrollHeight;
      lucide.createIcons();
    }

    function generateMedicalAnswer(query) {
      const lower = query.toLowerCase();
      let match = medicalKnowledge.find(entry => entry.triggers.some(t => lower.includes(t)));

      if (match) {
        appendChatMessage('bot', match.reply, match.suggestedMeds);
      } else {
        appendChatMessage(
          'bot',
          `Based on your query: "${query}", keep the patient rested in a safe, elevated environment. For hydration and standard relief, we can dispatch our flood trauma kit or rehydration pack right away.`,
          [
            { name: "ORS & Electrolyte Rehydration Pack", price: 4.50 },
            { name: "Flood Trauma & Suture Kit", price: 24.00 }
          ]
        );
      }
    }
